/* ---------- easing ---------- */
function bez(x1,y1,x2,y2){
  const cx=3*x1, bx=3*(x2-x1)-cx, ax=1-cx-bx, cy=3*y1, by=3*(y2-y1)-cy, ay=1-cy-by;
  const sx=t=>((ax*t+bx)*t+cx)*t, sy=t=>((ay*t+by)*t+cy)*t, dsx=t=>(3*ax*t+2*bx)*t+cx;
  return x=>{ if(x<=0) return 0; if(x>=1) return 1; let t=x;
    for(let i=0;i<8;i++){const d=sx(t)-x; if(Math.abs(d)<1e-6) break; const dd=dsx(t); if(Math.abs(dd)<1e-6) break; t-=d/dd;}
    let lo=0,hi=1; for(let i=0;i<30 && Math.abs(sx(t)-x)>1e-6;i++){ if(sx(t)<x) lo=t; else hi=t; t=(lo+hi)/2; }
    return sy(t); };
}
const E = bez(0.16,1,0.3,1);           // easing da marca
const IO = bez(0.65,0,0.35,1);         // in-out para wipes
const clamp=(v,a=0,b=1)=>Math.min(b,Math.max(a,v));
const P=(t,a,b)=>clamp((t-a)/(b-a));
const lerp=(a,b,p)=>a+(b-a)*p;
const backOut=p=>{const c=1.9;return 1+(c+1)*Math.pow(p-1,3)+c*Math.pow(p-1,2);};

/* ---------- utilidades de render ---------- */
function place(el, key, s, ax, ay, sx, sy){
  const m=META[key];
  const tx = sx - (m.ix + ax*m.iw)*s, ty = sy - (m.iy + ay*m.ih)*s;
  el.style.transform = `translate(${tx.toFixed(2)}px,${ty.toFixed(2)}px) scale(${s.toFixed(5)})`;
}
function show(el, o){ el.style.opacity = o; el.style.visibility = o<=0.001?'hidden':'visible'; }
function revealWords(ws, t, start, step, dur, out){
  ws.forEach((w,i)=>{
    const p = E(P(t, start+i*step, start+i*step+dur));
    let o = p, y = (1-p)*0.42;
    if (out){ const q = IO(P(t,out[0],out[1])); o *= 1-q; y -= q*0.25; }
    w.style.opacity = o.toFixed(3);
    w.style.transform = `translateY(${y.toFixed(3)}em)`;
    w.style.filter = p<1 ? `blur(${((1-p)*10).toFixed(2)}px)` : 'none';
  });
}
function fadeEl(el, t, a, b, out, dy=10){
  const p=E(P(t,a,b)); let o=p, y=(1-p)*dy;
  if(out){ const q=IO(P(t,out[0],out[1])); o*=1-q; }
  el.style.opacity=o.toFixed(3); el.style.transform=`translateY(${y.toFixed(2)}px)`;
  el.style.visibility = o<=0.001?'hidden':'visible';
}
